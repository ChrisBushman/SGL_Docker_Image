#
#   SEGA SATURN make file for GNU C++
#

# macro
CC = gcc
LD = ld
RM = - /bin/rm -f

# source_program
OBJS		= examp.o

TARGET		= sl.cof
LNK_FILE	= sl.lnk
MAP_FILE	= sl.map
MAKEFILE	= Makefile

# directory
SRC_DIR		= .
OBJ_DIR		= .
SEGA_INC	= ../SATURN/SGL/INC
SEGA_LIB	= ../SATURN/SGL/LIB
GNU_DIR		= ../CYGNUS/LIB/SH2
LINK_DIR	= .

# option
CCFLAGS		= -O0 -m2 -g -c -I$(SEGA_INC)
LDFLAGS		= -T$(LNK_FILE) -Map $(MAP_FILE) -e ___Start

SLIBS		= $(SEGA_LIB)/libsgl.a $(GNU_DIR)/libgcc.a
GLIBS		= $(GNU_DIR)/libgcc.a
LIBS		= $(SLIBS) $(GLIBS)

$(TARGET) : $(OBJS) $(MAKEFILE) $(LDFILE)
	$(LD) $(LDFLAGS) $(OBJS) $(LIBS) -o $@

examp.o : examp.cpp
	$(CC) $(CCFLAGS) examp.cpp -o examp.o

clean:
	$(RM) $(OBJS) sl.*
